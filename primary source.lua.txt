        function obfuscateLua(source, antiDebug) {
            const bytes = [];
            for (let i = 0; i < source.length; i++) {
                bytes.push(source.charCodeAt(i));
            }

            const varArray = "_0x" + Math.floor(Math.random()*16777215).toString(16);
            const varStr = "_0x" + Math.floor(Math.random()*16777215).toString(16);
            const varByte = "_0x" + Math.floor(Math.random()*16777215).toString(16);

            let header = `--[[ Obfuscated with PRIMARY Lua Protect ]]--\n`;
            
            if (antiDebug) {
                header += `if pcall and not pcall(function() return game end) then end\n`;
            }

            const scriptBody = `local ${varArray} = {${bytes.join(',')}}
local ${varStr} = ""
for _, ${varByte} in ipairs(${varArray}) do
    ${varStr} = ${varStr} .. string.char(${varByte})
end
return (loadstring or load)(${varStr})()`;

            return header + scriptBody;
        }

        function obfuscateJS(source) {
            const encoded = btoa(source);
            return `/* Obfuscated with PRIMARY JS Engine */
eval(atob("${encoded}"));`;
        }

        function runObfuscation() {
            const source = document.getElementById('sourceInput').value.trim();
            if (!source) {
                alert("Please put your source code first!");
                return;
            }

            const antiDebug = document.getElementById('chkAntiDebug').checked;
            let result = "";

            if (currentLang === 'lua') {
                result = obfuscateLua(source, antiDebug);
            } else {
                result = obfuscateJS(source);
            }

            document.getElementById('outputArea').value = result;
            logConsole(`✅ Code Obfuscated Successfully! [Mode: ${currentLang.toUpperCase()}]`);
        }

        function downloadSourceFile() {
            const source = document.getElementById('sourceInput').value;
            if (!source) {
                alert("Please input or paste your source code first!");
                return;
            }

            const fileName = currentLang === 'lua' ? "primary source.lua" : "primary source.js";
            triggerDownload(source, fileName);
            logConsole(`📥 Downloaded source file: ${fileName}`);
        }

        function downloadObfuscatedFile() {
            const output = document.getElementById('outputArea').value;
            if (!output) {
                alert("Please obfuscate your code first before downloading!");
                return;
            }

            const fileName = currentLang === 'lua' ? "primary obfuscated.lua" : "primary obfuscated.js";
            triggerDownload(output, fileName);
            logConsole(`📥 Downloaded obfuscated file: ${fileName}`);
        }

        function triggerDownload(content, fileName) {
            const blob = new Blob([content], { type: "text/plain;charset=utf-8" });
            const link = document.createElement("a");
            link.href = URL.createObjectURL(blob);
            link.download = fileName;
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);
        }

        function copyOutput() {
            const output = document.getElementById('outputArea');
            if (!output.value) return;
            output.select();
            document.execCommand('copy');
            logConsole("📋 Output copied to clipboard!");
        }

        function clearAll() {
            document.getElementById('sourceInput').value = '';
            document.getElementById('outputArea').value = '';
            logConsole("Cleared source and output.");
        }

        const canvas = document.getElementById('bgCanvas');
        const ctx = canvas.getContext('2d');
        let width, height;

        function resize() {
            width = canvas.width = window.innerWidth;
            height = canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resize);
        resize();

        const blobs = [
            { x: width * 0.2, y: height * 0.3, r: 180, vx: 1, vy: 0.8, color: 'rgba(255, 0, 128, 0.25)' },
            { x: width * 0.8, y: height * 0.7, r: 220, vx: -0.8, vy: -1, color: 'rgba(0, 255, 204, 0.2)' },
            { x: width * 0.5, y: height * 0.5, r: 150, vx: 0.6, vy: -0.6, color: 'rgba(153, 0, 255, 0.25)' }
        ];

        function animateBg() {
            ctx.clearRect(0, 0, width, height);

            blobs.forEach(b => {
                b.x += b.vx;
                b.y += b.vy;

                if (b.x < 0 || b.x > width) b.vx *= -1;
                if (b.y < 0 || b.y > height) b.vy *= -1;

                const grad = ctx.createRadialGradient(b.x, b.y, 0, b.x, b.y, b.r);
                grad.addColorStop(0, b.color);
                grad.addColorStop(1, 'rgba(0,0,0,0)');

                ctx.fillStyle = grad;
                ctx.beginPath();
                ctx.arc(b.x, b.y, b.r, 0, Math.PI * 2);
                ctx.fill();
            });

            requestAnimationFrame(animateBg);
        }
        animateBg();
